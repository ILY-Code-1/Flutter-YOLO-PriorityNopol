class DetectionRecord {
  - id : String
  - plateNumber : String
  - imageData : String
  - detectedAt : DateTime
  - confidence : double
  + isAsset() : bool
  + formattedDate() : String
}

  Police
  FireTruck
}

enum ConfidenceLevel {
  High
  Medium
  Low
}

class DetectionHistory {
  - selectedCategory : String

}
